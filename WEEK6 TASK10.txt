class Result {

    /*
     * Complete the 'circularPalindromes' function below.
     *
     * The function is expected to return an INTEGER_ARRAY.
     * The function accepts STRING s as parameter.
     */

    public static List<Integer> circularPalindromes(String s) {
    // Write your code here
    

    

        List<Integer> result = new ArrayList<>();

        int n = s.length();

        // Generate all rotations
        for (int k = 0; k < n; k++) {

            String rotated = s.substring(k) + s.substring(0, k);

            int maxLength = 1;

            // Check all possible substrings
            for (int i = 0; i < n; i++) {
                for (int j = i; j < n; j++) {

                    String sub = rotated.substring(i, j + 1);

                    // Check if substring is palindrome
                    int left = 0;
                    int right = sub.length() - 1;
                    boolean palindrome = true;

                    while (left < right) {

                        if (sub.charAt(left) != sub.charAt(right)) {
                            palindrome = false;
                            break;
                        }

                        left++;
                        right--;
                    }

                    if (palindrome) {
                        maxLength = Math.max(maxLength, sub.length());
                    }
                }
            }

            result.add(maxLength);
        }

        return result;
    }
}
    



public class Solution {
    public static void main(String[] args) throws IOException {
        BufferedReader bufferedReader = new BufferedReader(new InputStreamReader(System.in));

        int n = Integer.parseInt(bufferedReader.readLine().trim());

        String s = bufferedReader.readLine();

        List<Integer> result = Result.circularPalindromes(s);

        bufferedWriter.write(
            result.stream()
                .map(Object::toString)

            + "\n"
        );

    }
